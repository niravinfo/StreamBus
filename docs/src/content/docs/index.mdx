import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Message publishing" icon="paper-airplane">
    Publish plain objects without manual serialization. Super-stream routing keys are compile-time checked.
  </Card>
  <Card title="Consumer groups" icon="group">
    Receive endpoints with first-class, required consumer-group identity — a super
    stream's partitions scale across instances; different groups are independent cursors.
  </Card>
  <Card title="Retry & errors" icon="warning">
    Global and per-endpoint retry schedules, and dead-letter destinations — an error
    stream, an AMQP queue, or your own callback.
  </Card>
  <Card title="Explicit config" icon="settings">
    No binders, no JSON discovery. Every option is a configurator method, validated once at registration.
  </Card>
</CardGrid>
